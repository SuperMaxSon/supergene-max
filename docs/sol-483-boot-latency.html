-- ⚠ 전제가 틀렸다. 이 주석은 "483 이하에는 time 이 없다"고 적혀 있었으나,
--    Q1b 실측 결과 483~484 에도 ALL_LOAD_* · INTRO_FB_START_FINISH 의 data.time 이 있다

    SELECT
        IF(LOWER(JSON_VALUE(data, '$.isNewDevice')) = 'true', '신규기기', '재방문기기') AS seg,
        event,
        CAST(JSON_VALUE(data, '$.time') AS INT64) AS t
    FROM `game-log-359704.raw.solitaire_city_journey`
    WHERE log_date BETWEEN '2026-09-02' AND '2026-09-04'
      AND (is_dev = FALSE OR is_dev IS NULL)

      AND event IN (
          'FIRST_LOG',              -- main.js 진입
          'INTRO_FB_START',         -- FBInstant.initializeAsync 호출 직전
          'INTRO_FB_START_FINISH',  -- initializeAsync 완료

          'ALL_LOAD_INTRO_START',   -- runSceneImmediate 시작
          'ALL_LOAD_INTRO_FINISH'   -- 씬 전환 완료
      )
)

    seg                                    AS `기기`,
    event                                  AS `로그`,
    COUNT(*)                               AS `로그수`,
    CAST(ROUND(AVG(t)) AS INT64)           AS `평균_ms`,

    APPROX_QUANTILES(t, 100)[OFFSET(90)]   AS `p90_ms`
FROM src
WHERE t BETWEEN 0 AND 600000
GROUP BY 1, 2
ORDER BY `기기`, `p50_ms`

Q6-V — 자 검증 (솔리테어 · log_time 역산 vs data.time) ⚠ 아직 미실행 
-- ⚠ 아직 돌리지 않았다. 위 점선은 이 검증 없이 올라간 값이다.
-- 솔리테어는 data.time 과 log_time 을 둘 다 갖고 있어 두 자의 오차를 직접 잴 수 있다.
-- 판정: 오차_p50 이 ±150ms 안, 절대오차_p90 이 500ms 안이면 점선을 그대로 쓴다.
--       넘으면 점선을 1010 한 점으로 되돌리고 그 사실을 각주에 박는다.
WITH src AS (
    SELECT
        CONCAT(CAST(player_id AS STRING), '_', CAST(logincount_total AS STRING)) AS sk,
        event, log_time,
        CAST(JSON_VALUE(data, '$.time') AS INT64) AS t
    FROM `game-log-359704.raw.solitaire_city_journey`
    WHERE log_date = '2026-09-03'
      AND (is_dev = FALSE OR is_dev IS NULL) AND player_id > 0 AND country = 'US'
      AND logincount_total IS NOT NULL
      AND event IN (
          'INTRO_BUNDLE','INTRO_BUNDLE_FINISH',
          'INTRO_SIGNED_PLAYER_INFO','INTRO_SIGNED_PLAYER_INFO_FINISH',
          'INTRO_FB_FRIEND','INTRO_FB_FRIEND_FINISH',
          'INTRO_FB_ENTRY_POINT','INTRO_FB_ENTRY_POINT_FINISH',
          'INTRO_FB_ENTRY_DATA','INTRO_FB_ENTRY_DATA_FINISH',
          'INTRO_FB_SUBSCRIBE_BOT','INTRO_FB_SUBSCRIBE_BOT_FINISH',
          'INTRO_OBJECT_POOL','INTRO_OBJECT_POOL_FINISH',
          'INTRO_LOGIN','1000_LOGIN_COMPLETE','1010_LOAD_COMPLETE'
      )
),
anchor AS (
    SELECT sk, DATETIME_SUB(log_time, INTERVAL t MILLISECOND) AS t0
    FROM src
    WHERE event = '1010_LOAD_COMPLETE' AND t BETWEEN 0 AND 600000
    QUALIFY ROW_NUMBER() OVER (PARTITION BY sk ORDER BY log_time) = 1
),
j AS (
    SELECT s.event, s.t AS client_ms,
           DATETIME_DIFF(s.log_time, a.t0, MILLISECOND) AS server_ms
    FROM src s JOIN anchor a USING (sk)
    WHERE s.t IS NOT NULL
)
SELECT
    event                                                         AS `로그`,
    COUNT(*)                                                      AS `표본`,
    APPROX_QUANTILES(client_ms, 100)[OFFSET(50)]                  AS `client_p50_ms`,
    APPROX_QUANTILES(server_ms, 100)[OFFSET(50)]                  AS `server_p50_ms`,
    APPROX_QUANTILES(server_ms - client_ms, 100)[OFFSET(50)]      AS `오차_p50_ms`,
    APPROX_QUANTILES(ABS(server_ms - client_ms), 100)[OFFSET(90)] AS `절대오차_p90_ms`
FROM j
GROUP BY 1
ORDER BY `client_p50_ms`
