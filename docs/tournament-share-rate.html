WITH ours AS (
  SELECT 'ours_438plus' AS app,
    COUNTIF(event = '2100_GAMEPLAY_START')            AS plays,
    COUNTIF(event = '3110_TOURNAMENT_CREATE_SUCCESS') AS create_succ,
    COUNTIF(event = '3210_TOURNAMENT_SHARE_SUCCESS')  AS share_succ,
    COUNTIF(event = '3100_TOURNAMENT_CREATE')         AS create_try,
    COUNTIF(event = '3200_TOURNAMENT_SHARE')          AS share_try
  FROM `game-log-359704.raw.solitaire_city_journey`
  WHERE log_date BETWEEN '2026-08-08' AND '2026-08-11' AND client_version >= 438
),
mahjong AS (
  SELECT 'mahjong' AS app,
    COUNTIF(event = '2100_GAMEPLAY_START')            AS plays,
    COUNTIF(event = '3110_TOURNAMENT_CREATE_SUCCESS') AS create_succ,
    COUNTIF(event = '3210_TOURNAMENT_SHARE_SUCCESS')  AS share_succ,
    COUNTIF(event = '3100_TOURNAMENT_CREATE')         AS create_try,
    COUNTIF(event = '3200_TOURNAMENT_SHARE')          AS share_try
  FROM `game-log-359704.raw.mahjong`
  WHERE log_date BETWEEN '2026-08-08' AND '2026-08-11'
)
SELECT app, plays,
  ROUND(100 * SAFE_DIVIDE(create_try,  plays), 2) AS create_try_pct,
  ROUND(100 * SAFE_DIVIDE(share_try,   plays), 2) AS share_try_pct,
  ROUND(100 * SAFE_DIVIDE(create_succ, plays), 3) AS create_succ_pct,
  ROUND(100 * SAFE_DIVIDE(share_succ,  plays), 3) AS share_succ_pct,
  ROUND(100 * SAFE_DIVIDE(create_succ + share_succ, plays), 3) AS total_viral_pct
FROM (SELECT * FROM ours UNION ALL SELECT * FROM mahjong)
ORDER BY app;
빌드 귀인 판정 (NRU/RU 분해 · 동일 날짜창) 
SELECT
  client_version AS build,
  CASE WHEN logincount_total = 1 THEN 'NRU' ELSE 'RU' END AS user_type,
  COUNTIF(event = '3100_TOURNAMENT_CREATE')         AS try_cnt,
  COUNTIF(event = '3110_TOURNAMENT_CREATE_SUCCESS') AS success_cnt,
  ROUND(100 * SAFE_DIVIDE(
    COUNTIF(event = '3110_TOURNAMENT_CREATE_SUCCESS'),
    COUNTIF(event = '3100_TOURNAMENT_CREATE')), 2)  AS rate_pct
FROM `game-log-359704.raw.solitaire_city_journey`
WHERE log_date BETWEEN '2026-08-03' AND '2026-08-05'
  AND client_version IN (409, 415, 416)
  AND event IN ('3100_TOURNAMENT_CREATE', '3110_TOURNAMENT_CREATE_SUCCESS')
GROUP BY build, user_type
HAVING try_cnt >= 100
ORDER BY user_type, build;

SELECT
